# 设置/重绑定 UI 实现计划(KitRebindButton / KitRebindList)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development(推荐)或 superpowers:executing-plans 逐任务实现。步骤用 checkbox(`- [ ]`)跟踪。

**Goal:** 新增 off-facade UI 组件层 `ui/`:`KitRebindButton`(单动作重绑定:点击捕获 + 检测拦截冲突 + 复位)+ `KitRebindList`(多动作列表 + 全部复位);给 `InputManager` 加 `action_for_event` 冲突查询。audio/locale 接线放 demo 示范。

## 返回当前占用该事件的动作名;among 空=扫全部动作,否则只扫子集;无占用返回 ""。
## 复用 _event_to_dict 做结构相等比较(键盘按 physical_keycode 等)。供重绑 UI 判冲突。
func action_for_event(event: InputEvent, among := PackedStringArray()) -> String:
	var target := _event_to_dict(event)
	if target.is_empty():
		return ""
	var scan: Array = Array(among) if among.size() > 0 else InputMap.get_actions()
	for action in scan:
		if not InputMap.has_action(action):
			continue
		for ev in InputMap.action_get_events(action):
			if _event_to_dict(ev) == target:
				return String(action)
	return ""
```

```gdscript
class_name KitRebindButton
extends Button
## KitRebindButton —— 单动作键位重绑定按钮(off-facade 可复用 UI 组件)。
## 点击进捕获态 → 玩家按新键 → 冲突检测(检测+拦截)→ 落到 InputManager.rebind_action
## (替换 InputMap + 持久化 Settings)。只管 UI + 捕获 + 冲突判定;重绑/复位/持久化
## 复用注入的 InputManager(默认 CoreKit.input,可注入 standalone 做 headless 单测)。
##
## 零门面耦合:不引 CoreKit.log(告警走 push_warning);对 CoreKit.input 的默认解析是
## 消费侧便利,测试注入 standalone InputManager 绕开。捕获 Key/MouseButton/JoypadButton
## (对齐 InputManager 编码支持,忽略 joypad motion);Esc 取消。

## 重绑成功。
signal rebound(action: String, event: InputEvent)
## 重绑失败。reason ∈ "conflict:<other>" / "unsupported" / "cancelled"。
signal rebind_failed(action: String, reason: String)

const CAPTURE_PROMPT := "按任意键…(Esc 取消)"

var _action: String = ""
var _conflict_scope: PackedStringArray = PackedStringArray()
var _input = null                      # untyped:未注入时运行期解析 CoreKit.input
var _capturing: bool = false


func _ready() -> void:
	set_process_input(false)
	pressed.connect(begin_capture)


## 绑定动作。input_manager 省略则运行期用 CoreKit.input;conflict_scope 空=冲突扫全部。
func setup(action: String, conflict_scope := PackedStringArray(), input_manager = null) -> void:
	_action = action
	_conflict_scope = conflict_scope
	_input = input_manager
	refresh()


## 进捕获态:提示文字 + 抢焦点 + 开输入监听。
func begin_capture() -> void:
	if _capturing:
		return
	_capturing = true
	text = CAPTURE_PROMPT
	set_process_input(true)
	grab_focus()


## 编程/UI 取消:退捕获态、刷新、发 cancelled。
func cancel_capture() -> void:
	if not _capturing:
		return
	_finish_capture()
	refresh()
	rebind_failed.emit(_action, "cancelled")


## 可测种子:处理一个候选事件。Esc→取消;不支持→unsupported;冲突→conflict:<other>
## 且不改绑定;否则 rebind_action → rebound。返回是否成功重绑。
func capture_event(event: InputEvent) -> bool:
	if event is InputEventKey and event.physical_keycode == KEY_ESCAPE:
		_finish_capture()
		refresh()
		rebind_failed.emit(_action, "cancelled")
		return false
	if not _is_capturable(event):
		_finish_capture()
		rebind_failed.emit(_action, "unsupported")
		return false
	var mgr = _resolve_input()
	var other := String(mgr.action_for_event(event, _conflict_scope))
	if other != "" and other != _action:
		_finish_capture()
		rebind_failed.emit(_action, "conflict:%s" % other)
		return false
	if not mgr.rebind_action(_action, event):
		_finish_capture()
		rebind_failed.emit(_action, "unsupported")
		return false
	_finish_capture()
	refresh()
	rebound.emit(_action, event)
	return true


## 刷新文字为当前绑定的可读名;无绑定显「未绑定」。
func refresh() -> void:
	var mgr = _resolve_input()
	var ev: InputEvent = mgr.get_action_event(_action)
	text = ev.as_text() if ev != null else "未绑定"


## 复位到工程默认(InputManager 内部会发 action_rebound)并刷新。
func reset() -> void:
	_resolve_input().reset_action(_action)
	refresh()


func _input(event: InputEvent) -> void:
	if not _capturing:
		return
	if event is InputEventKey and event.physical_keycode == KEY_ESCAPE and event.pressed:
		capture_event(event)
		accept_event()
		return
	if _is_capturable(event):
		capture_event(event)
		accept_event()


func _finish_capture() -> void:
	_capturing = false
	set_process_input(false)


func _is_capturable(event: InputEvent) -> bool:
	if event is InputEventKey:
		return event.pressed and not event.echo
	if event is InputEventMouseButton:
		return event.pressed
	if event is InputEventJoypadButton:
		return event.pressed
	return false


func _resolve_input():
	return _input if _input != null else CoreKit.input
```

```gdscript
class_name KitRebindList
extends VBoxContainer
## KitRebindList —— 多动作重绑定列表(off-facade 可复用 UI 组件)。
## 吃 [{action, label}, ...],每行建 [Label | KitRebindButton | 复位],底部「全部复位」。
## 各行按钮的 conflict_scope = 全部行动作(冲突只在可重绑集合内判)。转发子按钮信号。

signal rebound(action: String, event: InputEvent)
signal rebind_failed(action: String, reason: String)

var _input = null
var _buttons: Array = []                       # KitRebindButton 列表(与行同序)
var _row_actions: PackedStringArray = PackedStringArray()


## rows = [{action, label}, ...]。input_manager 省略则各按钮用 CoreKit.input。
func setup(rows: Array, input_manager = null) -> void:
	_input = input_manager
	_clear()
	_row_actions = PackedStringArray()
	for row in rows:
		_row_actions.append(String(row.get("action", "")))
	for row in rows:
		_add_row(String(row.get("action", "")), String(row.get("label", row.get("action", ""))))
	var reset_all_btn := Button.new()
	reset_all_btn.text = "全部复位"
	reset_all_btn.pressed.connect(_on_reset_all)
	add_child(reset_all_btn)


## 刷新所有子按钮(复位后 / 外部变更后)。
func refresh_all() -> void:
	for btn in _buttons:
		btn.refresh()


func _add_row(action: String, label_text: String) -> void:
	var row := HBoxContainer.new()
	var label := Label.new()
	label.text = label_text
	label.size_flags_horizontal = Control.SIZE_EXPAND_FILL
	row.add_child(label)
	var btn := KitRebindButton.new()
	row.add_child(btn)
	btn.setup(action, _row_actions, _input)
	btn.rebound.connect(func(a, e): rebound.emit(a, e))
	btn.rebind_failed.connect(func(a, r): rebind_failed.emit(a, r))
	var reset_btn := Button.new()
	reset_btn.text = "复位"
	reset_btn.pressed.connect(btn.reset)
	row.add_child(reset_btn)
	add_child(row)
	_buttons.append(btn)


## 只复位本列表的行(不用 reset_all,避免复位列表外动作)。
func _on_reset_all() -> void:
	var mgr = _input if _input != null else CoreKit.input
	for action in _row_actions:
		mgr.reset_action(action)
	refresh_all()


func _clear() -> void:
	for child in get_children():
		child.queue_free()
	_buttons.clear()
```

- [ ] **Step 3: 创建 `test/test_rebind.gd`(完整内容)**

```gdscript
extends SceneTree
## KitRebindButton / KitRebindList headless 断言测试。经 capture_event 种子脱离真实
## 输入测核心逻辑;注入 standalone InputManager(临时动作 + _snapshot_defaults)。
## 运行:godot --headless --path . -s res://test/test_rebind.gd
##
## 加载注:input_manager.gd 运行期引用全局 autoload CoreKit,故 load() 于 _initialize;
## KitRebindButton/List(class_name)一并 load(),避免类作用域 preload 的解析期坑。
## 隔离:临时动作用完 InputMap.erase_action 清除。组件注入 im,从不触及 CoreKit.input。

var RebindButtonScript
var RebindListScript
var InputManagerScript

const A_JUMP := "__rb_jump__"
const A_FIRE := "__rb_fire__"


func _key(kc: int, pressed := true) -> InputEventKey:
	var ev := InputEventKey.new()
	ev.physical_keycode = kc
	ev.pressed = pressed
	return ev


func _initialize() -> void:
	RebindButtonScript = load("res://addons/godot-core-kit/ui/rebind_button.gd")
	RebindListScript = load("res://addons/godot-core-kit/ui/rebind_list.gd")
	InputManagerScript = load("res://addons/godot-core-kit/systems/input_manager.gd")
	var failures := 0

	# 临时动作:jump=SPACE,fire=G
	InputMap.add_action(A_JUMP)
	InputMap.action_add_event(A_JUMP, _key(KEY_SPACE, false))
	InputMap.add_action(A_FIRE)
	InputMap.action_add_event(A_FIRE, _key(KEY_G, false))
	var im = InputManagerScript.new()
	im._snapshot_defaults()

	var scope := PackedStringArray([A_JUMP, A_FIRE])
	var btn = RebindButtonScript.new()
	btn.setup(A_JUMP, scope, im)
	var rebounds := []
	var fails := []
	btn.rebound.connect(func(a, e): rebounds.append(a))
	btn.rebind_failed.connect(func(a, r): fails.append(r))

	# 1. 正常重绑 jump -> F
	var ok1: bool = btn.capture_event(_key(KEY_F))
	var jump_events := InputMap.action_get_events(A_JUMP)
	if not ok1 or jump_events.size() != 1 or jump_events[0].physical_keycode != KEY_F:
		print("FAIL 1: 正常重绑 jump 应变 F,ok=%s events=%s" % [ok1, jump_events])
		failures += 1
	if rebounds != [A_JUMP]:
		print("FAIL 1: 应发一次 rebound(jump),实际 %s" % [rebounds])
		failures += 1

	# 2. 冲突拦截:jump 重绑到 G(fire 占用)-> 拒绝、不改
	var ok2: bool = btn.capture_event(_key(KEY_G))
	if ok2:
		print("FAIL 2: 冲突应返回 false")
		failures += 1
	if fails != ["conflict:%s" % A_FIRE]:
		print("FAIL 2: 应发 rebind_failed conflict:%s,实际 %s" % [A_FIRE, fails])
		failures += 1
	if InputMap.action_get_events(A_JUMP)[0].physical_keycode != KEY_F:
		print("FAIL 2: 冲突后 jump 绑定应保持 F 不变")
		failures += 1

	# 3. 不支持类型
	fails.clear()
	var ok3: bool = btn.capture_event(InputEventMouseMotion.new())
	if ok3 or fails != ["unsupported"]:
		print("FAIL 3: 不支持类型应 rebind_failed unsupported,实际 %s" % [fails])
		failures += 1

	# 4. Esc 取消
	fails.clear()
	var ok4: bool = btn.capture_event(_key(KEY_ESCAPE))
	if ok4 or fails != ["cancelled"]:
		print("FAIL 4: Esc 应 rebind_failed cancelled,实际 %s" % [fails])
		failures += 1
	if InputMap.action_get_events(A_JUMP)[0].physical_keycode != KEY_F:
		print("FAIL 4: 取消后 jump 绑定应不变(F)")
		failures += 1

	# 5. reset 回默认 SPACE
	btn.reset()
	if InputMap.action_get_events(A_JUMP)[0].physical_keycode != KEY_SPACE:
		print("FAIL 5: reset 应回默认 SPACE")
		failures += 1

	# 6. action_for_event(among 作用域,避开 ui_accept 默认占 Space)
	if im.action_for_event(_key(KEY_G), scope) != A_FIRE:
		print("FAIL 6: action_for_event(G) 应为 fire")
		failures += 1
	if im.action_for_event(_key(KEY_G), PackedStringArray([A_JUMP])) != "":
		print("FAIL 6: among 子集不含 fire 时应为空")
		failures += 1
	if im.action_for_event(InputEventScreenTouch.new(), scope) != "":
		print("FAIL 6: 不支持事件 action_for_event 应为空")
		failures += 1
	btn.free()

	# 7. KitRebindList:2 行 + 全部复位 + 转发信号
	im.reset_action(A_JUMP)  # 干净起步:jump 回 SPACE
	var list = RebindListScript.new()
	list.setup([{"action": A_JUMP, "label": "Jump"}, {"action": A_FIRE, "label": "Fire"}], im)
	if list._buttons.size() != 2:
		print("FAIL 7: 列表应建 2 个重绑按钮,实际 %d" % list._buttons.size())
		failures += 1
	var list_rebounds := []
	list.rebound.connect(func(a, e): list_rebounds.append(a))
	list._buttons[0].capture_event(_key(KEY_F))  # 经首个按钮重绑 jump -> F(不冲突)
	if list_rebounds != [A_JUMP]:
		print("FAIL 7: 列表应转发 rebound(jump),实际 %s" % [list_rebounds])
		failures += 1
	list._on_reset_all()
	if InputMap.action_get_events(A_JUMP)[0].physical_keycode != KEY_SPACE:
		print("FAIL 7: 全部复位应把 jump 回 SPACE")
		failures += 1
	list.free()

	# 清理
	im.free()
	InputMap.erase_action(A_JUMP)
	InputMap.erase_action(A_FIRE)

	if failures == 0:
		print("test_rebind: ALL PASS")
		quit(0)
	else:
		print("test_rebind: %d FAILURE(S)" % failures)
		quit(1)

```markdown
# 键位重绑定 UI(KitRebindButton / KitRebindList)

> `class_name KitRebindButton` / `KitRebindList` · `ui/rebind_button.gd` / `ui/rebind_list.gd` · off-facade 可复用 UI 组件(不挂门面)

## 职责

可复用的游戏内键位重绑定 UI。`KitRebindButton` 一个动作一个按钮:点击 → 按新键捕获 → 冲突检测(检测+拦截)→ 落到 `CoreKit.input.rebind_action`(替换 InputMap + 持久化 Settings)。`KitRebindList` 吃动作清单一次性建列表 + 全部复位。组件只管 UI + 捕获 + 冲突判定,重绑/复位/持久化复用 InputManager。audio 音量、locale 切换是一行绑定的琐碎活,自己拼即可(见 demo)。

## 公开 API

### KitRebindButton(extends Button)

| 成员 | 说明 |
|---|---|
| `setup(action, conflict_scope := PackedStringArray(), input_manager = null)` | 绑定动作;`input_manager` 省略用 `CoreKit.input`;`conflict_scope` 空=冲突扫全部动作,否则只在子集内判。 |
| `begin_capture()` | 进捕获态(点击自动触发):提示「按任意键…」+ 抢焦点。 |
| `capture_event(event) -> bool` | 处理候选事件(真实输入与测试共用的种子)。Esc→取消;不支持类型→`unsupported`;冲突→`conflict:<other>` 且不改绑定;否则重绑成功。 |
| `cancel_capture()` | 取消捕获(编程/UI)。 |
| `refresh()` | 文字=当前绑定 `as_text()`,无绑定显「未绑定」。 |
| `reset()` | 复位到工程默认 + 刷新。 |
| `signal rebound(action, event)` / `signal rebind_failed(action, reason)` | reason ∈ `conflict:<other>` / `unsupported` / `cancelled`。 |

捕获类型:Key / MouseButton / JoypadButton(忽略摇杆轴);Esc 取消。

### KitRebindList(extends VBoxContainer)

| 成员 | 说明 |
|---|---|
| `setup(rows, input_manager = null)` | `rows=[{action, label}, ...]`;每行 `[Label \| KitRebindButton \| 复位]` + 底部「全部复位」;各按钮 `conflict_scope`=全部行动作。 |
| `refresh_all()` | 刷新所有子按钮。 |
| `signal rebound / rebind_failed` | 转发各子按钮信号。 |

## 依赖

- 消费 `CoreKit.input`(`InputManager`)的 `action_for_event`/`rebind_action`/`reset_action`/`get_action_event`。**零门面耦合**:告警走 `push_warning`,对 `CoreKit.input` 的默认解析可注入 standalone InputManager 绕开(headless 单测)。off-facade:不挂门面,`.new()`+`add_child` 即用。

## 用法示例

```gdscript
# 一次性列表(推荐)
var list := KitRebindList.new()
add_child(list)
list.setup([
    {"action": "move_left",  "label": "左移"},
    {"action": "move_right", "label": "右移"},
    {"action": "jump",       "label": "跳跃"},
])
list.rebind_failed.connect(func(a, r): if r.begins_with("conflict:"): $Toast.show("已被占用"))

# 单个按钮
var btn := KitRebindButton.new()
add_child(btn)
btn.setup("jump", PackedStringArray(["jump", "attack"]))

# audio / locale 是一行绑定,自己拼(无需组件):
$VolumeSlider.value_changed.connect(func(v): CoreKit.audio.set_volume("master", v))
$LocaleOption.item_selected.connect(func(i): CoreKit.i18n.set_locale(locales[i]))
```

## 注意事项

- **冲突范围**:`conflict_scope` 决定冲突只在哪些动作间判——`KitRebindList` 自动传全部行(避免「Space 撞 ui_accept」误报)。单用 `KitRebindButton` 时自己传可重绑集合。
- **同键重绑自身**:把动作重绑到它当前已有的键 → 放行(成功、等价 no-op)。
- **持久化**:重绑经 `InputManager.rebind_action` 写 Settings section `input`;下次启动 `bind_settings` 恢复。复位清除覆盖。
- **v1 范围外**:冲突交换(只做拦截)、摇杆轴当按键、多绑定槽、组合键。
