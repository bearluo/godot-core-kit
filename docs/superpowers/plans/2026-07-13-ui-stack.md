# UI 栈导航 (KitUiStack + KitUiScreen) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给 godot-core-kit 增加一个 off-facade 通用 UI 栈导航器 `KitUiStack` + 可选屏组件 `KitUiScreen`,管一叠 UI 屏的 push/pop/replace/clear、路由输入、驱动生命周期。

**Architecture:** 两个 `class_name` 类库落在现有 `addons/godot-core-kit/ui/`。`KitUiStack extends Control`(全屏容器,push 进来的屏为其子节点);被盖屏 `process_mode=DISABLED` 一刀切停用,栈顶保持活动,返回键经宿主 `_unhandled_input` 兜底。`KitUiScreen extends Node` 为可选组件,挂进屏根即得生命周期信号(entered/exited/covered/revealed)+ 转发方法(pop/push/replace)+ 回指 `stack`。零框架依赖。

**Tech Stack:** Godot 4.7,纯 GDScript,Web 导出兼容。headless SceneTree 测试(`godot --headless -s res://test/*.gd`)。

```gdscript
class_name KitUiScreen
extends Node
## 可选 UI 屏组件。屏幕想要生命周期回调 / 公共操作就挂它做子节点;宿主 KitUiStack
## 在 push 时注入 stack 回指并驱动其信号。零依赖,可脱离框架单独用。
## 生命周期信号由 KitUiStack 驱动;prev/next 为对手屏节点名,无则 ""。

signal entered(prev_name: String)   ## 本屏成为栈顶、激活
signal exited(next_name: String)    ## 本屏被 pop、即将移除
signal covered()                    ## 被上层 push 盖住、被 process_mode 停用
signal revealed()                   ## 上层 pop 后重新成为栈顶、恢复活动

## 宿主回指;push 时由 KitUiStack 注入。不标注 KitUiStack 类型(破 class_name 循环引用)。
var stack


func pop() -> void:
	if stack != null:
		stack.pop()


func push(scene_or_node, opts := {}) -> Control:
	if stack != null:
		return stack.push(scene_or_node, opts)
	return null


func replace(scene_or_node, opts := {}) -> Control:
	if stack != null:
		return stack.replace(scene_or_node, opts)
	return null


## 从任意(深层)子节点往上找到所属屏的 KitUiScreen 组件;无则 null。
static func of(node: Node) -> KitUiScreen:
	var n: Node = node
	while n != null:
		for c in n.get_children():
			if c is KitUiScreen:
				return c
		n = n.get_parent()
	return null
```

- [ ] **Step 2: 写 `kit_ui_stack.gd`**

```gdscript
class_name KitUiStack
extends Control
## 通用 UI 栈导航宿主(off-facade 类库)。挂到任意父节点(某 CanvasLayer/Control)下,
## push/pop/replace/clear 管一叠 UI 屏(LIFO)。push 进来的屏是本节点子节点 →
## 只管加进来的、不碰你手动摆的。被盖屏 process_mode=DISABLED 一刀切停用(键鼠/动作/
## tick 全惰性);栈顶保持活动、用自己的输入回调;返回键经本节点 _unhandled_input 兜底。
## 屏若挂 KitUiScreen 组件则收生命周期信号 + 得公共方法。零框架依赖。

signal screen_pushed(screen: Control)
signal screen_popped(screen: Control)
signal emptied()

## 返回键动作名;为空 StringName 则关闭"返回键弹栈"。
@export var cancel_action: StringName = &"ui_cancel"

## 栈条目:{screen, backdrop, own, cancellable, prev_mode, prev_visible}。栈底->栈顶。
var _stack: Array[Dictionary] = []


func _ready() -> void:
	set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	mouse_filter = Control.MOUSE_FILTER_IGNORE


## 压一屏。scene_or_node: PackedScene(实例化)或现成 Control。返回屏实例;非法 → null。
func push(scene_or_node, opts := {}) -> Control:
	var screen := _instantiate(scene_or_node)
	if screen == null:
		return null
	var prev_name := ""
	if not _stack.is_empty():
		var top_entry: Dictionary = _stack[-1]
		prev_name = String((top_entry["screen"] as Control).name)
		_cover(top_entry, opts.get("hide_below", true))
	_mount_new(screen, opts, prev_name)
	return screen


## 弹栈顶。空栈 → push_warning + no-op。
func pop() -> void:
	if _stack.is_empty():
		push_warning("KitUiStack: 栈空,pop 忽略")
		return
	var next_name := ""
	if _stack.size() >= 2:
		next_name = String((_stack[-2]["screen"] as Control).name)
	var top_screen: Control = _stack[-1]["screen"]
	_teardown_top(next_name)
	if not _stack.is_empty():
		_reveal(_stack[-1])
	screen_popped.emit(top_screen)
	if _stack.is_empty():
		emptied.emit()


## 换顶(栈深不变):拆旧顶但不 reveal 下层,再把新屏盖到同一下层。空栈时等价纯 push。
func replace(scene_or_node, opts := {}) -> Control:
	if _stack.is_empty():
		return push(scene_or_node, opts)
	var screen := _instantiate(scene_or_node)
	if screen == null:
		return null
	_teardown_top(String(screen.name))
	var prev_name := ""
	if not _stack.is_empty():
		prev_name = String((_stack[-1]["screen"] as Control).name)
	_mount_new(screen, opts, prev_name)
	return screen


## 从顶到底逐个弹空,末发 emptied。
func clear() -> void:
	if _stack.is_empty():
		return
	while not _stack.is_empty():
		var top_screen: Control = _stack[-1]["screen"]
		_teardown_top("")
		screen_popped.emit(top_screen)
	emptied.emit()


func top() -> Control:
	return _stack[-1]["screen"] if not _stack.is_empty() else null


func depth() -> int:
	return _stack.size()


func is_empty() -> bool:
	return _stack.is_empty()


## 返回键兜底:仅当没人消费该 unhandled 输入时才触发(栈顶自定义返回则不弹)。
func _unhandled_input(event: InputEvent) -> void:
	if _stack.is_empty() or cancel_action == &"":
		return
	if event.is_action_pressed(cancel_action) and _stack[-1]["cancellable"]:
		pop()
		get_viewport().set_input_as_handled()


## --- 内部 ---

## 归一 push 参数为 Control 实例;PackedScene 则 instantiate。非法 → push_error + null。
func _instantiate(scene_or_node) -> Control:
	if scene_or_node is PackedScene:
		var inst: Node = (scene_or_node as PackedScene).instantiate()
		if inst is Control:
			return inst
		push_error("KitUiStack.push: PackedScene 实例化结果非 Control,拒绝")
		if inst != null:
			inst.free()
		return null
	if scene_or_node is Control:
		return scene_or_node
	push_error("KitUiStack.push: 参数须为 PackedScene 或 Control,拒绝")
	return null


## 盖住一屏:记录原 process_mode/visible → DISABLED 停用 →(hide)隐藏 → covered()。
func _cover(entry: Dictionary, hide: bool) -> void:
	var screen: Control = entry["screen"]
	entry["prev_mode"] = screen.process_mode
	entry["prev_visible"] = screen.visible
	screen.process_mode = Node.PROCESS_MODE_DISABLED
	if hide:
		screen.visible = false
	var comp := KitUiScreen.of(screen)
	if comp != null:
		comp.covered.emit()


## 揭开一屏:还原 process_mode/visible → revealed()。
func _reveal(entry: Dictionary) -> void:
	var screen: Control = entry["screen"]
	screen.process_mode = entry.get("prev_mode", Node.PROCESS_MODE_INHERIT)
	screen.visible = entry.get("prev_visible", true)
	var comp := KitUiScreen.of(screen)
	if comp != null:
		comp.revealed.emit()


## 拆栈顶(不 reveal 下层):exited(next_name) → 移挡层 → 按 own free/detach 屏。
func _teardown_top(next_name: String) -> void:
	var entry: Dictionary = _stack.pop_back()
	var screen: Control = entry["screen"]
	var comp := KitUiScreen.of(screen)
	if comp != null:
		comp.exited.emit(next_name)
	if entry["backdrop"] != null:
		(entry["backdrop"] as Node).queue_free()
	remove_child(screen)
	if entry["own"]:
		screen.queue_free()


## 挂新屏为栈顶:(模态)插挡层 → add_child → 记条目 → 注入 stack + entered() → screen_pushed。
func _mount_new(screen: Control, opts: Dictionary, prev_name: String) -> void:
	var backdrop: ColorRect = null
	if not opts.get("hide_below", true):
		backdrop = ColorRect.new()
		backdrop.color = opts.get("dim", Color(0, 0, 0, 0))
		backdrop.mouse_filter = Control.MOUSE_FILTER_STOP
		add_child(backdrop)
		backdrop.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	add_child(screen)
	_stack.append({
		"screen": screen,
		"backdrop": backdrop,
		"own": opts.get("own", true),
		"cancellable": opts.get("cancellable", true),
		"prev_mode": Node.PROCESS_MODE_INHERIT,
		"prev_visible": true,
	})
	var comp := KitUiScreen.of(screen)
	if comp != null:
		comp.stack = self
		comp.entered.emit(prev_name)
	screen_pushed.emit(screen)
```

```gdscript
extends SceneTree
## KitUiStack + KitUiScreen headless 断言。宿主入树(root.add_child + await frame)后
## 程序化 push/pop/replace/clear;屏用代码建的最小 Control(挂 KitUiScreen 子节点)。
## 运行:godot --headless --path . -s res://test/test_ui_stack.gd
## 前置:先跑过一次 --import(class_name 缓存),否则 is KitUiScreen 解析不到。

var StackScript
var ScreenScript
var journal: Array = []


## 建一屏:Control 根 + KitUiScreen 子组件,组件信号写进 journal。
func _mk(nm: String) -> Control:
	var s := Control.new()
	s.name = nm
	var comp = ScreenScript.new()
	comp.name = "KitUiScreen"
	s.add_child(comp)
	comp.entered.connect(func(prev: String) -> void: journal.append([nm, "entered", prev]))
	comp.exited.connect(func(nx: String) -> void: journal.append([nm, "exited", nx]))
	comp.covered.connect(func() -> void: journal.append([nm, "covered"]))
	comp.revealed.connect(func() -> void: journal.append([nm, "revealed"]))
	return s


func _initialize() -> void:
	StackScript = load("res://addons/godot-core-kit/ui/kit_ui_stack.gd")
	ScreenScript = load("res://addons/godot-core-kit/ui/kit_ui_screen.gd")
	var failures := 0

	var stack = StackScript.new()
	root.add_child(stack)
	await process_frame   # _initialize 期 root 未入树,补一帧触发 _ready

	# 1. push/depth/top
	var a1 = _mk("A"); stack.push(a1)
	var b1 = _mk("B"); stack.push(b1)
	var c1 = _mk("C"); var pushed = stack.push(c1)
	if stack.depth() != 3 or stack.top() != c1 or pushed != c1:
		print("FAIL 1: push 后 depth 应 3、top 应 C,depth=%d" % stack.depth()); failures += 1

	# 2. pop 顺序 + 空栈守卫
	stack.pop(); stack.pop()
	if stack.depth() != 1 or stack.top() != a1:
		print("FAIL 2: pop 两次后 depth 应 1、top 应 A,depth=%d" % stack.depth()); failures += 1
	stack.pop()
	stack.pop()   # 空栈再 pop → no-op 不崩
	if stack.depth() != 0:
		print("FAIL 2: 空栈 pop 应 no-op,depth=%d" % stack.depth()); failures += 1

	# 3. replace 换顶不改深 + 旧顶被 free
	var a3 = _mk("A3"); stack.push(a3)
	var b3 = _mk("B3"); stack.push(b3)
	var c3 = _mk("C3"); stack.replace(c3)
	if stack.depth() != 2 or stack.top() != c3:
		print("FAIL 3: replace 后 depth 应 2、top 应 C3,depth=%d" % stack.depth()); failures += 1
	await process_frame
	if is_instance_valid(b3):
		print("FAIL 3: replace 应 free 旧顶 B3"); failures += 1
	stack.clear()

	# 4. clear + emptied
	var emptied_count := [0]
	stack.emptied.connect(func() -> void: emptied_count[0] += 1)
	stack.push(_mk("X")); stack.push(_mk("Y"))
	stack.clear()
	if stack.depth() != 0 or emptied_count[0] != 1:
		print("FAIL 4: clear 后 depth 应 0、emptied 一次,depth=%d cnt=%d" % [stack.depth(), emptied_count[0]]); failures += 1

	# 5. hide_below=true 隐藏 + 停用下层
	var a5 = _mk("A5"); stack.push(a5)
	var b5 = _mk("B5"); stack.push(b5)   # 默认 hide_below=true
	if a5.visible != false or a5.can_process() != false:
		print("FAIL 5: 默认 push 应隐藏 + 停用下层,visible=%s can_process=%s" % [a5.visible, a5.can_process()]); failures += 1
	stack.pop()
	if a5.visible != true or a5.can_process() != true:
		print("FAIL 5: pop 后下层应还原,visible=%s can_process=%s" % [a5.visible, a5.can_process()]); failures += 1
	stack.clear()

	# 6. 模态 hide_below=false 保留可见 + 停用 + 插挡层
	var a6 = _mk("A6"); stack.push(a6)
	var b6 = _mk("B6"); stack.push(b6, {"hide_below": false, "dim": Color(0, 0, 0, 0.5)})
	var backdrop = stack._stack[-1]["backdrop"]
	if a6.visible != true or a6.can_process() != false:
		print("FAIL 6: 模态下层应可见但停用,visible=%s can_process=%s" % [a6.visible, a6.can_process()]); failures += 1
	if backdrop == null or backdrop.mouse_filter != Control.MOUSE_FILTER_STOP or backdrop.color != Color(0, 0, 0, 0.5):
		print("FAIL 6: 模态应插 STOP 挡层 + dim 色"); failures += 1
	stack.pop()
	await process_frame
	if a6.can_process() != true or is_instance_valid(backdrop):
		print("FAIL 6: pop 模态后下层应恢复、挡层应被 free"); failures += 1
	stack.clear()

	# 7. 生命周期信号 + 顺序
	journal.clear()
	var a7 = _mk("A"); stack.push(a7)   # A entered ""
	var b7 = _mk("B"); stack.push(b7)   # A covered, B entered "A"
	stack.pop()                          # B exited "A", A revealed
	var expect7 = [["A", "entered", ""], ["A", "covered"], ["B", "entered", "A"], ["B", "exited", "A"], ["A", "revealed"]]
	if journal != expect7:
		print("FAIL 7: 生命周期序列不符,实际 %s" % [journal]); failures += 1
	stack.clear()

	# 8. own 归属
	var keep := Control.new(); keep.name = "Keep"
	stack.push(keep, {"own": false})
	stack.pop()
	if not is_instance_valid(keep):
		print("FAIL 8: own=false 应仅 detach 不 free"); failures += 1
	if is_instance_valid(keep):
		keep.free()
	var drop = _mk("Drop"); stack.push(drop)   # own 默认 true
	stack.pop()
	await process_frame
	if is_instance_valid(drop):
		print("FAIL 8: own=true pop 后应被 free"); failures += 1

	# 9. KitUiScreen.of() 解析
	var s9 = _mk("S9")
	var deep := Button.new()
	s9.add_child(deep)
	stack.push(s9)
	if ScreenScript.of(deep) == null or ScreenScript.of(deep) != s9.get_node("KitUiScreen"):
		print("FAIL 9: of() 应从深层子节点找到本屏组件"); failures += 1
	var bare := Control.new()   # 无组件屏
	if ScreenScript.of(bare) != null:
		print("FAIL 9: 无组件屏 of() 应为 null"); failures += 1
	bare.free()
	stack.clear()

	# 10. 回指注入 + ui.pop() 等价
	var s10 = _mk("S10"); stack.push(s10)
	var comp10 = s10.get_node("KitUiScreen")
	if comp10.stack != stack:
		print("FAIL 10: push 应注入 stack 回指"); failures += 1
	stack.push(_mk("S10b"))
	var comp10b_depth_before := stack.depth()
	var comp10b = stack.top().get_node("KitUiScreen")
	comp10b.pop()   # 经组件转发弹自己
	if stack.depth() != comp10b_depth_before - 1:
		print("FAIL 10: ui.pop() 应等价宿主 pop(),depth 未 -1"); failures += 1
	stack.clear()

	# 11. cancel 弹栈 + cancellable=false 不弹
	var s11 = _mk("S11"); stack.push(s11)
	var ev := InputEventAction.new()
	ev.action = stack.cancel_action
	ev.pressed = true
	stack._unhandled_input(ev)
	if stack.depth() != 0:
		print("FAIL 11: cancel 应弹栈顶,depth=%d" % stack.depth()); failures += 1
	var s11b = _mk("S11b"); stack.push(s11b, {"cancellable": false})
	stack._unhandled_input(ev)
	if stack.depth() != 1:
		print("FAIL 11: cancellable=false 不应被 cancel 弹出,depth=%d" % stack.depth()); failures += 1
	stack.clear()

	# 12. 非法 push
	var before12 := stack.depth()
	var r1 = stack.push(Resource.new())
	var r2 = stack.push(Node.new())
	if r1 != null or r2 != null or stack.depth() != before12:
		print("FAIL 12: 非法 push 应返回 null 且不入栈,depth=%d" % stack.depth()); failures += 1

	stack.free()

	if failures == 0:
		print("test_ui_stack: ALL PASS")
		quit(0)
	else:
		print("test_ui_stack: %d FAILURE(S)" % failures)
		quit(1)
```
