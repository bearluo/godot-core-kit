# Settings 子系统 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现玩家偏好分区键值子系统 Settings(写透落盘 `user://settings.json`、变更信号、损坏降级)。

**Architecture:** 见 spec `docs/superpowers/specs/2026-07-02-settings-design.md`。单文件 Node;门面接线(`CoreKit.settings` 属性)不在本任务,留给集成任务。

**Tech Stack:** Godot 4.7 / 纯 GDScript / headless `-s` 断言测试

## Global Constraints

- 分支:全部工作在 `feat/framework-v0.4` 上(从 main 切出);线性历史,不得产生 merge commit。
- 纯 GDScript、Tab 缩进;无 C#、无 GDExtension、无第三方依赖;命令行 `godot` 已在 PATH。
- Web 导出兼容:禁用线程/平台专属 API。
- 依赖纪律:子系统间引用仅限总纲 spec 依赖表(本子系统仅运行期 `CoreKit.log.error`);初始化期诊断用 `push_warning`/`push_error`。
- 测试纪律(`-s` 避坑手册):`-s` 主脚本不得在类作用域 preload 任何引用 `CoreKit` 全局的脚本(`_initialize()` 内运行时 `load()`);untyped 值的下游变量显式标注类型(不用 `:=`);测试无论成败必须到达 `quit()`。
- 运行测试用硬超时模板(PowerShell,单测 60s 上限):

```powershell
$p = Start-Process godot -ArgumentList '--headless','--path','.','-s','res://test/test_settings.gd' -RedirectStandardOutput "$env:TEMP\gck-out.txt" -RedirectStandardError "$env:TEMP\gck-err.txt" -PassThru -NoNewWindow; if (-not $p.WaitForExit(60000)) { $p.Kill(); 'TIMEOUT(疑似测试报错未达 quit)' }; Get-Content "$env:TEMP\gck-out.txt"
```

### Task 1: Settings 核心

**Files:**
- Create: `addons/godot-core-kit/settings.gd`
- Test: `test/test_settings.gd`

**Interfaces:**
- Consumes: `CoreKit.log.error(msg: String)`(落盘失败时;门面迁移任务已交付)。
- Produces: `signal value_changed(section: String, key: String, value: Variant)`;`var settings_path: String`(默认 `user://settings.json`,测试注入接缝——spec 里的 `const SETTINGS_PATH` 在实现中放宽为 var,语义不变);`get_value(section, key, default=null) -> Variant`;`set_value(section, key, value) -> void`;`has_value(section, key) -> bool`;`erase_value(section, key) -> void`;`erase_section(section) -> void`;接缝 `_encode(data: Dictionary) -> String`、`_decode(text: String) -> Dictionary`、`_load_from_disk() -> void`。InputManager/AudioManager 任务将经 `bind_settings()` 拿到本实例并调用 `get_value`/`set_value`/`erase_value`。

- [ ] **Step 1: 写失败测试 test/test_settings.gd**

```gdscript
extends SceneTree
## Settings headless 断言测试(直接实例化脚本,不依赖门面接线)。
## 运行:godot --headless --path . -s res://test/test_settings.gd
##
## 加载注:settings.gd 运行期引用全局 autoload 标识符 CoreKit(落盘失败时
## CoreKit.log.error),故不在类作用域 preload,延迟到 _initialize() 内 load()
## (此时 autoload 已注册)。settings_path 接缝重定向到一次性测试文件,
## 不碰真实 user://settings.json(那是 CoreKit 门面实例在用的真文件)。

var SettingsScript

const TEST_PATH := "user://__test_settings__.json"


func _make():
	var s = SettingsScript.new()
	s.settings_path = TEST_PATH
	return s


func _initialize() -> void:
	SettingsScript = load("res://addons/godot-core-kit/settings.gd")
	var failures := 0

	# 起点干净(上次异常退出可能残留)
	if FileAccess.file_exists(TEST_PATH):
		DirAccess.remove_absolute(TEST_PATH)

	var s = _make()

	# 1. 未设值返回 default
	if s.get_value("audio", "music") != null:
		print("FAIL 1: 未设值应返回 null")
		failures += 1
	if s.get_value("audio", "music", 0.5) != 0.5:
		print("FAIL 1: 未设值应返回显式 default")
		failures += 1

	# 2. set → get 往返(标量与容器)
	s.set_value("audio", "music", 0.8)
	if s.get_value("audio", "music") != 0.8:
		print("FAIL 2: float 往返失败")
		failures += 1
	s.set_value("gfx", "fullscreen", true)
	if s.get_value("gfx", "fullscreen") != true:
		print("FAIL 2: bool 往返失败")
		failures += 1
	s.set_value("meta", "tag", "hero")
	if s.get_value("meta", "tag") != "hero":
		print("FAIL 2: String 往返失败")
		failures += 1

	# 3. 信号:同值不广播;新值广播一次且参数正确
	var calls := []
	s.value_changed.connect(func(section: String, key: String, value) -> void:
		calls.append([section, key, value]))
	s.set_value("audio", "music", 0.8)  # 同值
	if calls.size() != 0:
		print("FAIL 3: 同值 set 不应广播,收到 %d 次" % calls.size())
		failures += 1
	s.set_value("audio", "music", 0.3)
	if calls.size() != 1 or calls[0] != ["audio", "music", 0.3]:
		print("FAIL 3: 新值 set 应广播一次且参数正确,收到 %s" % [calls])
		failures += 1

	# 4. has_value / erase_value / erase_section 语义与空操作安全
	if not s.has_value("audio", "music"):
		print("FAIL 4: has_value 应为 true")
		failures += 1
	s.erase_value("audio", "music")
	if s.has_value("audio", "music"):
		print("FAIL 4: erase_value 后 has_value 应为 false")
		failures += 1
	if calls.size() != 2 or calls[1] != ["audio", "music", null]:
		print("FAIL 4: erase_value 应广播 value=null")
		failures += 1
	s.erase_value("audio", "music")   # 再删:空操作,不广播
	s.erase_section("no_such_section")  # 空操作
	if calls.size() != 2:
		print("FAIL 4: 空操作不应广播")
		failures += 1
	s.erase_section("gfx")
	if s.has_value("gfx", "fullscreen"):
		print("FAIL 4: erase_section 后键应消失")
		failures += 1

	# 5. _decode 损坏输入;_decode(_encode(x)) 往返
	if s._decode("not valid json {{{") != {}:
		print("FAIL 5: 非法 JSON 应返回 {}")
		failures += 1
	if s._decode("42") != {}:
		print("FAIL 5: 根非 Dictionary 应返回 {}")
		failures += 1
	var round_tripped: Dictionary = s._decode(s._encode({"a": {"b": "c"}}))
	if round_tripped.get("a", {}).get("b") != "c":
		print("FAIL 5: _encode/_decode 往返失败,得到 %s" % [round_tripped])
		failures += 1
	# 非 Dictionary 的 section 值在 _decode 时被丢弃
	var dropped: Dictionary = s._decode('{"good": {"k": 1}, "bad": 5}')
	if dropped.has("bad") or not dropped.has("good"):
		print("FAIL 5: 非 Dictionary section 应被丢弃,得到 %s" % [dropped])
		failures += 1

	# 6. 磁盘往返:重建实例读盘一致(注:JSON 往返 int 会变 float,
	#    顶层标量 == 比较 42 == 42.0 为 true,断言仍成立)
	s.set_value("save", "score", 42)
	var s2 = _make()
	s2._load_from_disk()
	if s2.get_value("save", "score") != 42:
		print("FAIL 6: 磁盘往返失败,得到 %s" % [s2.get_value("save", "score")])
		failures += 1
	if s2.get_value("meta", "tag") != "hero":
		print("FAIL 6: 磁盘往返丢键")
		failures += 1
	s2.free()

	# 7. 坏文件降级:预写非法 JSON → 空配置且可正常覆盖写
	var f := FileAccess.open(TEST_PATH, FileAccess.WRITE)
	f.store_string("corrupted[[[")
	f.close()
	var s3 = _make()
	s3._load_from_disk()
	if s3.get_value("meta", "tag") != null:
		print("FAIL 7: 坏文件应降级为空配置")
		failures += 1
	s3.set_value("meta", "tag", "again")
	var s4 = _make()
	s4._load_from_disk()
	if s4.get_value("meta", "tag") != "again":
		print("FAIL 7: 坏文件被 set 覆盖后应可正常读回")
		failures += 1
	s3.free()
	s4.free()

	# 清理
	s.free()
	if FileAccess.file_exists(TEST_PATH):
		DirAccess.remove_absolute(TEST_PATH)

	if failures == 0:
		print("test_settings: ALL PASS")
		quit(0)
	else:
		print("test_settings: %d FAILURE(S)" % failures)
		quit(1)
```

- [ ] **Step 3: 实现 addons/godot-core-kit/settings.gd**

```gdscript
extends Node
## Settings —— 玩家偏好分区键值,写透落盘 user://settings.json。
##
## 定位:设置 ≠ 存档。音量/键位/语言这类"跨存档、装完游戏就该记住"的键值
## 落这里,不走 SaveManager(不出现在存档槽列表、不被删档误删)。
## 由 CoreKit 门面构造并经 CoreKit.settings 访问;InputManager/AudioManager
## 经 bind_settings() 注入本实例做持久化。
##
## 注意:JSON 往返会把 int 变 float(与 SaveManager 同款,JSON 规范决定),
## 取整用 int(get_value(...))。value 限 JSON 可表达类型(bool/int/float/
## String/Array/Dictionary),调用方自律,不做运行时校验。

## 任何 set_value/erase_* 实际改变了值之后广播(值未变化的 set 不广播);
## erase_value 广播 value=null;erase_section 广播一次 (section, "", null)。
signal value_changed(section: String, key: String, value: Variant)

## 落盘路径。var 而非 const:测试注入接缝(重定向到一次性文件),语义不变。
var settings_path: String = "user://settings.json"

## 内存态,形状 {section(String): {key(String): value}}。
var _data: Dictionary = {}


func _ready() -> void:
	_load_from_disk()


func get_value(section: String, key: String, default: Variant = null) -> Variant:
	if not _data.has(section):
		return default
	var sec: Dictionary = _data[section]
	if not sec.has(key):
		return default
	return sec[key]


func set_value(section: String, key: String, value: Variant) -> void:
	if _data.has(section) and _data[section].has(key) and _data[section][key] == value:
		return
	if not _data.has(section):
		_data[section] = {}
	_data[section][key] = value
	_save_to_disk()
	value_changed.emit(section, key, value)


func has_value(section: String, key: String) -> bool:
	return _data.has(section) and _data[section].has(key)


func erase_value(section: String, key: String) -> void:
	if not has_value(section, key):
		return
	_data[section].erase(key)
	if _data[section].is_empty():
		_data.erase(section)
	_save_to_disk()
	value_changed.emit(section, key, null)


func erase_section(section: String) -> void:
	if not _data.has(section):
		return
	_data.erase(section)
	_save_to_disk()
	value_changed.emit(section, "", null)


## --- 可测接缝 ---

func _encode(data: Dictionary) -> String:
	return JSON.stringify(data, "\t")


## 任何失败(JSON 非法、根不是 Dictionary)返回 {};非 Dictionary 的
## section 值在此丢弃(schema 不符视同损坏的局部)。
func _decode(text: String) -> Dictionary:
	var parsed = JSON.parse_string(text)
	if typeof(parsed) != TYPE_DICTIONARY:
		return {}
	var result: Dictionary = {}
	for section in parsed:
		if typeof(parsed[section]) == TYPE_DICTIONARY:
			result[section] = parsed[section]
	return result


func _load_from_disk() -> void:
	if not FileAccess.file_exists(settings_path):
		return
	var file := FileAccess.open(settings_path, FileAccess.READ)
	if file == null:
		push_warning("Settings: 无法读取 %s(错误码 %d),从空配置开始" % [settings_path, FileAccess.get_open_error()])
		return
	var text := file.get_as_text()
	file.close()
	if text.strip_edges().is_empty():
		return
	var decoded := _decode(text)
	if decoded.is_empty() and text.strip_edges() != "{}":
		push_warning("Settings: %s 内容损坏,从空配置开始(不删除原文件)" % settings_path)
	_data = decoded


func _save_to_disk() -> void:
	var file := FileAccess.open(settings_path, FileAccess.WRITE)
	if file == null:
		CoreKit.log.error("Settings: 无法写入 %s(错误码 %d),本次变更只在内存生效" % [settings_path, FileAccess.get_open_error()])
		return
	file.store_string(_encode(_data))
	file.close()
```
