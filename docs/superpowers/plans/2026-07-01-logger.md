# Logger 子系统 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 godot-core-kit 实现正式的日志子系统 `Log`(全局 autoload),替换脚手架里 `CoreKit.info()` 占位。

**Architecture:** 单文件 autoload(方案 A)。`log.gd` 一个文件承载:级别枚举、阈值过滤、格式化、彩色 `print_rich` 输出、`WARN`/`ERROR` 额外 `push_warning`/`push_error`。纯逻辑抽成 `_should_log` / `format_line` 两个可测接缝,headless 断言脚本直接实例化脚本测试(不依赖 autoload)。`Log` 先于 `CoreKit` 初始化,`CoreKit` 改为依赖 `Log`。

**Tech Stack:** Godot 4.7,纯 GDScript,Compatibility 渲染器。无第三方测试框架。

## Global Constraints

以下为全项目约束,每个任务隐含遵守:

- 引擎 Godot 4.7,`config_version=5`;纯 GDScript,**不含 C#**。
- **Web 导出兼容**:禁止文件 IO;只用 `print`/`print_rich`/`push_error`/`push_warning`/`Time.*` 等 Web 可用 API。
- autoload 顺序:**`Log` 必须声明/注册在 `CoreKit` 之前**(`CoreKit._ready()` 会调 `Log`)。
- 日志格式**逐字**:`[HH:MM:SS.mmm][LEVEL] message`,例 `[14:22:07.512][INFO] hello`。
- 级别有序:`enum Level { DEBUG, INFO, WARN, ERROR }`;默认阈值 `Level.DEBUG`。
- 颜色映射:`DEBUG`=gray、`INFO`=默认色(不包 color 标签)、`WARN`=yellow、`ERROR`=red。
- 调试器集成:`WARN`→额外 `push_warning(纯文本行)`;`ERROR`→额外 `push_error(纯文本行)`(传入不含 BBCode 的行)。
- 本子系统版本:`0.1.0` → `0.2.0`。
- 本机 Godot(验证用),下文命令记作 `$godot`:
  `D:\Program Files\Godot_v4.7-stable_mono_win64\Godot_v4.7-stable_mono_win64\Godot_v4.7-stable_mono_win64_console.exe`
  PowerShell 里先设 `$godot = "D:\Program Files\Godot_v4.7-stable_mono_win64\Godot_v4.7-stable_mono_win64\Godot_v4.7-stable_mono_win64_console.exe"`,再 `& $godot ...`。

- [ ] **Step 1: 写失败测试** `test/test_log.gd`

```gdscript
extends SceneTree
## Log headless 断言测试(不依赖 autoload,直接实例化脚本)。
## 运行:godot --headless -s res://test/test_log.gd

const LogScript := preload("res://addons/godot-core-kit/log.gd")


func _initialize() -> void:
	var failures := 0
	var logger = LogScript.new()  # 变量名避开内置 log() 函数

	# 1. format_line 结构:[HH:MM:SS.mmm][INFO] hello
	var line: String = logger.format_line(LogScript.Level.INFO, "hello")
	var re := RegEx.new()
	re.compile("^\\[\\d\\d:\\d\\d:\\d\\d\\.\\d\\d\\d\\]\\[INFO\\] hello$")
	if re.search(line) == null:
		print("FAIL format_line: 得到 '%s'" % line)
		failures += 1

	# 2. 默认阈值 DEBUG 放行 DEBUG
	if not logger._should_log(LogScript.Level.DEBUG):
		print("FAIL 默认阈值应放行 DEBUG")
		failures += 1

	# 3. set_level(WARN) 后过滤 DEBUG/INFO,放行 WARN/ERROR
	logger.set_level(LogScript.Level.WARN)
	if logger._should_log(LogScript.Level.DEBUG):
		print("FAIL WARN 阈值不应放行 DEBUG")
		failures += 1
	if logger._should_log(LogScript.Level.INFO):
		print("FAIL WARN 阈值不应放行 INFO")
		failures += 1
	if not logger._should_log(LogScript.Level.WARN):
		print("FAIL WARN 阈值应放行 WARN")
		failures += 1
	if not logger._should_log(LogScript.Level.ERROR):
		print("FAIL WARN 阈值应放行 ERROR")
		failures += 1

	# 4. get_level 反映 set_level
	if logger.get_level() != LogScript.Level.WARN:
		print("FAIL get_level 应为 WARN")
		failures += 1

	logger.free()

	if failures == 0:
		print("test_log: ALL PASS")
		quit(0)
	else:
		print("test_log: %d FAILURE(S)" % failures)
		quit(1)
```

- [ ] **Step 3: 写实现** `addons/godot-core-kit/log.gd`

```gdscript
extends Node
## Log —— godot-core-kit 全局日志单例(骨架首版)。
##
## 以 autoload "Log" 注册。四级分级(DEBUG/INFO/WARN/ERROR)+ 统一格式
## [HH:MM:SS.mmm][LEVEL] message + 全局阈值过滤(默认 DEBUG,运行时 set_level 可改)
## + print_rich 彩色输出。WARN/ERROR 额外走 push_warning/push_error 进编辑器
## "错误"面板并带调用栈。纯 GDScript、无文件 IO,支持 Web 导出。
##
## 用法:Log.info("msg")、Log.warn("x=%d" % x)、Log.set_level(Log.Level.WARN)

enum Level { DEBUG, INFO, WARN, ERROR }

const _LEVEL_NAMES := ["DEBUG", "INFO", "WARN", "ERROR"]
## 颜色映射:INFO 用 "" 表示默认色,不包 [color] 标签。
const _LEVEL_COLORS := ["gray", "", "yellow", "red"]

var _level: Level = Level.DEBUG


func set_level(level: Level) -> void:
	_level = level


func get_level() -> Level:
	return _level


func debug(message) -> void:
	_log(Level.DEBUG, message)


func info(message) -> void:
	_log(Level.INFO, message)


func warn(message) -> void:
	_log(Level.WARN, message)


func error(message) -> void:
	_log(Level.ERROR, message)


func _should_log(level: Level) -> bool:
	return level >= _level


## 返回格式化后的一行,不打印(供测试与内部复用)。
func format_line(level: Level, message) -> String:
	var unix := Time.get_unix_time_from_system()
	var ms := int(fmod(unix, 1.0) * 1000.0)
	var time_str := Time.get_time_string_from_system()
	return "[%s.%03d][%s] %s" % [time_str, ms, _LEVEL_NAMES[level], str(message)]


func _log(level: Level, message) -> void:
	if not _should_log(level):
		return
	var line := format_line(level, message)
	var color: String = _LEVEL_COLORS[level]
	if color.is_empty():
		print_rich(line)
	else:
		print_rich("[color=%s]%s[/color]" % [color, line])
	if level == Level.WARN:
		push_warning(line)
	elif level == Level.ERROR:
		push_error(line)
```

整体替换文件内容为:

```gdscript
@tool
extends EditorPlugin
## godot-core-kit 编辑器插件入口。
##
## 消费方启用本插件时,按顺序注册 autoload:先 Log(最底层、零依赖),
## 再 CoreKit(依赖 Log)。顺序决定初始化顺序,CoreKit._ready() 会用到 Log。
## 每项带 has_setting 幂等判断,避免与开发工程 project.godot 预声明重复注册。
## 禁用时逆序移除。

const AUTOLOADS := [
	["Log", "res://addons/godot-core-kit/log.gd"],
	["CoreKit", "res://addons/godot-core-kit/core_kit.gd"],
]


func _enter_tree() -> void:
	for entry in AUTOLOADS:
		var autoload_name: String = entry[0]
		var autoload_path: String = entry[1]
		if not ProjectSettings.has_setting("autoload/" + autoload_name):
			add_autoload_singleton(autoload_name, autoload_path)


func _exit_tree() -> void:
	for i in range(AUTOLOADS.size() - 1, -1, -1):
		var autoload_name: String = AUTOLOADS[i][0]
		if ProjectSettings.has_setting("autoload/" + autoload_name):
			remove_autoload_singleton(autoload_name)
```
