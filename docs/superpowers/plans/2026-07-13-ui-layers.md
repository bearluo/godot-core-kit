- [ ] **Step 1: 写 `systems/layer_manager.gd`**

```gdscript
extends Node
## CoreKit.layers —— 具名 CanvasLayer 管理器(第 13 子系统)。在本节点下懒建一组
## 具名 CanvasLayer,z 序由常量表定义;因本节点是 CoreKit autoload 子节点,这些层
## 及挂在其上的内容独立于当前场景渲染、跨 change_scene 存活。零跨系统依赖。
##
## 保留带(本管理器不创建,仅登记常量供引用):
##   CONSOLE=128   —— DevConsole 自建自持(console/dev_console.gd)
##   TRANSITION=4096 —— SceneManager 淡入淡出遮罩自建自持(systems/scene_manager.gd)
## 新增自定义层的 z 应落在 SYSTEM(60)与 CONSOLE(128)之间以免盖过控制台/转场。

## --- z 序常量(权威单一数据源;游戏引用而非猜数字)---
const BACKGROUND := -10   ## 世界底下(视差/天幕)
const HUD := 10           ## 常驻血条/小地图
const WINDOW := 20        ## 游戏内面板
const POPUP := 30         ## 弹窗
const MODAL := 40         ## 模态
const NOTIFY := 50        ## Toast/成就弹
const SYSTEM := 60        ## Loading/暂停幕
const CONSOLE := 128      ## 保留带:DevConsole 现值(不创建)
const TRANSITION := 4096  ## 保留带:SceneManager 转场遮罩现值(不创建)

## 内置具名层 → z。register_layer 往这里加自定义项;console/transition 故意不在内。
var _z_map: Dictionary = {
	&"background": BACKGROUND, &"hud": HUD, &"window": WINDOW, &"popup": POPUP,
	&"modal": MODAL, &"notify": NOTIFY, &"system": SYSTEM,
}
var _layers: Dictionary = {}   # StringName -> CanvasLayer(已懒建缓存)


## 取(懒建)具名层。未知名(未内置且未 register)→ push_error + null。
func layer(name: StringName) -> CanvasLayer:
	if _layers.has(name):
		return _layers[name]
	if not _z_map.has(name):
		push_error("CoreKit.layers: 未知层名 '%s'(需内置名或先 register_layer)" % name)
		return null
	var cl := CanvasLayer.new()
	cl.name = "Layer_%s" % name
	cl.layer = _z_map[name]
	add_child(cl)
	_layers[name] = cl
	return cl


## 把节点搬进具名层(从原父摘下再挂)。未知层名 → push_error + no-op。
func mount(node: Node, name: StringName) -> void:
	var cl := layer(name)
	if cl == null:
		return
	if node.get_parent() != null:
		node.get_parent().remove_child(node)
	cl.add_child(node)


## 清空某层(queue_free 全部子节点)。未建的层 → no-op。
func clear(name: StringName) -> void:
	if not _layers.has(name):
		return
	for c in (_layers[name] as CanvasLayer).get_children():
		c.queue_free()


## 登记自定义层名→z(懒建在下次 layer() 时)。重名 → push_warning + 拒;
## z 落入保留带(>= CONSOLE)→ push_warning 但仍登记(进阶用途)。
func register_layer(name: StringName, z: int) -> void:
	if _z_map.has(name):
		push_warning("CoreKit.layers: 层名 '%s' 已存在,拒绝重复登记" % name)
		return
	if z >= CONSOLE:
		push_warning("CoreKit.layers: 自定义层 '%s' z=%d 落入保留带(>=%d),可能盖过控制台/转场" % [name, z, CONSOLE])
	_z_map[name] = z


## 是否已登记该层名(内置或 register 过);不创建。
func has_layer(name: StringName) -> bool:
	return _z_map.has(name)
```

- [ ] **Step 2: 写 `test/test_layer_manager.gd`**

```gdscript
extends SceneTree
## LayerManager headless 断言。管理器入树(root.add_child + await frame)后
## 懒建/取具名 CanvasLayer、mount/clear/register_layer。零依赖,直接 load() 实例化。
## 运行:godot --headless --path . -s res://test/test_layer_manager.gd

var LayerScript


func _initialize() -> void:
	LayerScript = load("res://addons/godot-core-kit/systems/layer_manager.gd")
	var failures := 0

	var lm = LayerScript.new()
	root.add_child(lm)
	await process_frame

	# 1. 懒建 + 缓存 + z 值
	var hud = lm.layer("hud")
	if hud == null or not (hud is CanvasLayer) or hud.layer != 10:
		print("FAIL 1: layer('hud') 应是 layer=10 的 CanvasLayer"); failures += 1
	if lm.layer("hud") != hud:
		print("FAIL 1: 二次 layer('hud') 应返回同一实例(缓存)"); failures += 1
	if lm.layer("modal").layer != 40 or lm.layer("background").layer != -10:
		print("FAIL 1: modal 应 40、background 应 -10"); failures += 1

	# 2. 内置 z 序常量
	if lm.HUD != 10 or lm.WINDOW != 20 or lm.POPUP != 30 or lm.MODAL != 40 or lm.NOTIFY != 50 or lm.SYSTEM != 60 or lm.BACKGROUND != -10:
		print("FAIL 2: 内置 z 常量不符"); failures += 1

	# 3. 保留带常量 + 不创建
	if lm.CONSOLE != 128 or lm.TRANSITION != 4096:
		print("FAIL 3: 保留带常量应 CONSOLE=128 TRANSITION=4096"); failures += 1
	if lm.has_layer("console") or lm.has_layer("transition"):
		print("FAIL 3: 管理器不应登记 console/transition"); failures += 1
	if lm.layer("console") != null:
		print("FAIL 3: layer('console') 未知名应 null"); failures += 1

	# 4. mount 搬入 + 摘旧父
	var tmp := Node.new()
	root.add_child(tmp)
	var n := Control.new()
	tmp.add_child(n)
	lm.mount(n, "popup")
	if n.get_parent() != lm.layer("popup") or tmp.get_child_count() != 0:
		print("FAIL 4: mount 应把 n 从旧父搬进 popup 层"); failures += 1
	tmp.free()

	# 5. clear
	lm.mount(Control.new(), "notify")
	lm.mount(Control.new(), "notify")
	lm.clear("notify")
	await process_frame
	if lm.layer("notify").get_child_count() != 0:
		print("FAIL 5: clear 应清空 notify 层"); failures += 1

	# 6. register_layer + 懒建
	lm.register_layer("weather", -5)
	if not lm.has_layer("weather") or lm.layer("weather").layer != -5:
		print("FAIL 6: register_layer('weather',-5) 后应可建 layer=-5"); failures += 1

	# 7. register 重名拒
	lm.register_layer("hud", 999)
	if lm.layer("hud").layer != 10:
		print("FAIL 7: 重名 register 应被拒,hud 仍 10"); failures += 1

	# 8. 层挂在管理器下(常驻结构佐证)
	if lm.layer("hud").get_parent() != lm:
		print("FAIL 8: 具名层应是 LayerManager 的子节点"); failures += 1

	lm.free()

	if failures == 0:
		print("test_layer_manager: ALL PASS")
		quit(0)
	else:
		print("test_layer_manager: %d FAILURE(S)" % failures)
		quit(1)
```
